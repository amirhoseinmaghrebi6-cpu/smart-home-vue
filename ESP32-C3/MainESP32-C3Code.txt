// ==================== کتابخانه‌ها ====================
#include <WiFi.h>
#include <WiFiManager.h>
#include <PubSubClient.h>
#include <Preferences.h>
#include <esp_mac.h>

// ==================== تنظیمات محصول ====================
#define PRODUCT_NAME        "SmartHome-C3"
#define CONFIG_PORTAL_PASS  "smart1234"

// ✅ مقادیر پیش‌فرض MQTT (قابل تغییر توسط کاربر در WiFiManager)
// 🔹 در پروداکشن: این مقدار را به دامنه‌ی سرور خود تغییر دهید
// مثال: #define MQTT_SERVER_DEFAULT "mqtt.yourapp.ir"
#define MQTT_SERVER_DEFAULT "10.177.64.123"  // ← فعلاً آی‌پی لوکال شما
#define MQTT_PORT_DEFAULT   1883
#define MQTT_KEEPALIVE      60
#define MQTT_USE_SSL        false  // اگر سرور SSL دارد، true کنید و پورت را 8883 بگذارید

// ==================== 🔑 پین‌های امن برای ESP32-C3 SuperMini ====================
// ✅ GPIO 4,5 → رله‌ها / خروجی‌ها (قابل تغییر بر اساس سخت‌افزار)
// ✅ GPIO 3 → تاچ / دکمه‌ی فیزیکی
// ✅ GPIO 2 → LED وضعیت داخلی برد
const int CHANNEL_PINS[] = {4, 5, 6};
const int NUM_CHANNELS = sizeof(CHANNEL_PINS) / sizeof(CHANNEL_PINS[0]);

#define TOUCH_PIN     3
#define LED_STATUS    2

// ==================== تایم‌اوت‌ها (برای UX ساده‌تر) ====================
#define TOUCH_DEBOUNCE_MS       50
#define PAIRING_MIN_MS          8000      // حداقل ۸ ثانیه برای Pairing
#define PAIRING_MAX_MS          12000     // حداکثر ۱۲ ثانیه برای Pairing
#define RESET_HOLD_MS           20000     // بیش از ۲۰ ثانیه = Factory Reset
#define STATUS_PUBLISH_INTERVAL 10000     // ارسال وضعیت هر ۱۰ ثانیه
#define MQTT_RECONNECT_MIN      2000
#define MQTT_RECONNECT_MAX      30000

// ==================== متغیرهای سراسری ====================
Preferences prefs;
String deviceId;           // پایدار بر پایه‌ی مک آدرس
String mqttServer;         // پویا: از Preferences یا پیش‌فرض
int mqttPort;              // پویا: از Preferences یا پیش‌فرض
String pairedUserId;       // کاربر مالک دستگاه
String configPortalSsid;   // نام پورتال تنظیم وای‌فای

bool isPaired = false;
bool channelStatus[NUM_CHANNELS] = {false};

WiFiManager wifiManager;
WiFiClient espClient;
PubSubClient mqttClient(espClient);

// ✅ حالت‌های تاچ
enum TouchState { IDLE, HOLDING };
TouchState touchState = IDLE;
unsigned long touchStartTime = 0;
unsigned long lastTouchChange = 0;
int lastTouchRead = HIGH;

unsigned long lastMqttReconnectAttempt = 0;
unsigned long mqttReconnectDelay = MQTT_RECONNECT_MIN;
bool mqttConnected = false;
unsigned long lastStatusPublish = 0;

// ==================== توابع پیش‌اعلام ====================
void initHardware();
void initPreferences();
void initWiFiManager();
void initMQTT();
void handleTouch();
void handleMQTT();
bool attemptMqttConnect();
void onMqttMessage(char* topic, byte* payload, unsigned int length);
void publishStatus(bool force = false);
void executeCommandWithLog(int channelIndex, bool state);
void sendPairingRequest();
void factoryReset();
void blinkLED(int times, int duration);
String getDeviceMacString();
String generateUniqueId();
String buildJson(const char* devId, bool* channels, bool online, bool paired);
String buildPairingJson();
String extractField(const String& json, const String& key);

// ==================== SETUP ====================
void setup() {
  Serial.begin(115200);
  delay(2000);  // صبر برای باز شدن Serial Monitor
  
  Serial.println("\n🚀 " + String(PRODUCT_NAME) + " Starting...");
  
  initHardware();
  initPreferences();   // ← اول Preferences را بخوان (شامل mqttServer)
  initWiFiManager();   // ← سپس وای‌فای را کانکت کن
  initMQTT();          // ← در آخر MQTT را با مقادیر خوانده‌شده تنظیم کن
  
  Serial.println("✅ System Ready. Waiting for events...");
}

// ==================== LOOP ====================
void loop() {
  handleTouch();
  
  // ✅ مدیریت اتصال WiFi
  if (WiFi.status() != WL_CONNECTED) {
    static unsigned long lastReconnectAttempt = 0;
    static int reconnectAttempts = 0;
    const unsigned long RECONNECT_INTERVAL = 5000;
    const int MAX_RECONNECT_ATTEMPTS = 10;
    
    unsigned long now = millis();
    
    if (now - lastReconnectAttempt > RECONNECT_INTERVAL) {
      lastReconnectAttempt = now;
      reconnectAttempts++;
      
      Serial.printf("⚠️ WiFi disconnected (Attempt %d/%d)...\n", reconnectAttempts, MAX_RECONNECT_ATTEMPTS);
      blinkLED(1, 100);
      
      WiFi.disconnect(true);
      delay(100);
      WiFi.begin(WiFi.SSID().c_str(), WiFi.psk().c_str());
      
      unsigned long start = millis();
      while (millis() - start < 10000 && WiFi.status() != WL_CONNECTED) {
        delay(500);
      }
      
      if (WiFi.status() == WL_CONNECTED) {
        Serial.println("✅ WiFi reconnected successfully!");
        Serial.printf("🌐 New IP: %s\n", WiFi.localIP().toString().c_str());
        reconnectAttempts = 0;
        blinkLED(2, 200);
      } 
      else if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) {
        Serial.println("❌ Max reconnection attempts reached. Starting Config Portal...");
        blinkLED(5, 100);
        
        WiFiManager wm;
        wm.setConfigPortalBlocking(true);
        wm.setConfigPortalTimeout(120);
        
        String portalSsid = "Setup_" + getDeviceMacString();
        if (wm.autoConnect(portalSsid.c_str(), CONFIG_PORTAL_PASS)) {
          Serial.println("✅ WiFi reconfigured via portal!");
          reconnectAttempts = 0;
        } else {
          Serial.println("❌ Portal timeout. Restarting...");
          delay(2000);
          ESP.restart();
        }
      }
    }
    delay(100);
    return;
  }
  
  // ✅ مدیریت MQTT
  handleMQTT();
  
  // ✅ ارسال دوره‌ای وضعیت
  if (mqttConnected && millis() - lastStatusPublish > STATUS_PUBLISH_INTERVAL) {
    publishStatus(false);
    lastStatusPublish = millis();
  }
  
  delay(50);
}

// ==================== راه‌اندازی سخت‌افزار ====================
void initHardware() {
  pinMode(LED_STATUS, OUTPUT);
  digitalWrite(LED_STATUS, LOW);
  pinMode(TOUCH_PIN, INPUT_PULLUP);
  
  for (int i = 0; i < NUM_CHANNELS; i++) {
    pinMode(CHANNEL_PINS[i], OUTPUT);
    digitalWrite(CHANNEL_PINS[i], LOW);
    Serial.printf("⚙️ Channel %d → GPIO%d configured\n", i, CHANNEL_PINS[i]);
  }
  
  String macStr = getDeviceMacString();
  Serial.printf("🔌 %s (MAC: %s)\n", PRODUCT_NAME, macStr.c_str());
}

// ==================== Preferences (ذخیره‌سازی پایدار) ====================
void initPreferences() {
  prefs.begin("smarthome", false);
  
  // ✅ Device ID پایدار بر پایه‌ی مک آدرس (هرگز تغییر نمی‌کند)
  if (!prefs.isKey("deviceId")) {
    deviceId = generateUniqueId();
    prefs.putString("deviceId", deviceId);
    Serial.printf("🆔 New Device ID: %s\n", deviceId.c_str());
  } else {
    deviceId = prefs.getString("deviceId");
    Serial.printf("🆔 Existing Device ID: %s\n", deviceId.c_str());
  }
  
  // ✅ خواندن تنظیمات MQTT از Preferences (یا استفاده از پیش‌فرض)
  mqttServer = prefs.getString("mqtt_server", MQTT_SERVER_DEFAULT);
  mqttPort = prefs.getInt("mqtt_port", MQTT_PORT_DEFAULT);
  Serial.printf("🔌 MQTT Config: %s:%d\n", mqttServer.c_str(), mqttPort);
  
  // ✅ بارگذاری وضعیت Pairing
  pairedUserId = prefs.getString("pairedUserId", "");
  isPaired = prefs.getBool("isPaired", false);
  Serial.printf("🔗 Paired: %s | User: %s\n", 
                isPaired ? "YES" : "NO", 
                pairedUserId.isEmpty() ? "none" : pairedUserId.c_str());
}

// ✅ تولید ID یونیک و پایدار: فقط بر پایه‌ی مک آدرس (بدون رندوم)
String generateUniqueId() {
  uint8_t mac[6];
  esp_read_mac(mac, ESP_MAC_WIFI_STA);
  char macStr[13];
  sprintf(macStr, "%02X%02X%02X%02X%02X%02X", 
          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  return String(macStr);
}

String getDeviceMacString() {
  uint8_t mac[6];
  esp_read_mac(mac, ESP_MAC_WIFI_STA);
  char macStr[13];
  sprintf(macStr, "%02X%02X%02X%02X%02X%02X", 
          mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
  return String(macStr);
}

// ========== WiFiManager با پشتیبانی از تنظیم MQTT توسط کاربر ==========
void initWiFiManager() {
  configPortalSsid = "Setup_" + getDeviceMacString();
  
  // ✅ تنظیمات حیاتی برای ESP32-C3
  wifiManager.setConfigPortalBlocking(true);
  wifiManager.setConfigPortalTimeout(600);  // ۱۰ دقیقه فرصت
  
  // ✅ پارامترهای قابل تنظیم توسط کاربر در پورتال
  WiFiManagerParameter custom_mqtt_server("server", "MQTT Server", mqttServer.c_str(), 60);
  WiFiManagerParameter custom_mqtt_port("port", "MQTT Port", String(mqttPort).c_str(), 6);
  
  wifiManager.addParameter(&custom_mqtt_server);
  wifiManager.addParameter(&custom_mqtt_port);
  
  // ✅ کالبک‌ها برای فیدبک بصری
  wifiManager.setSaveParamsCallback([]() { 
    Serial.println("💾 WiFi + MQTT credentials saved"); 
    blinkLED(3, 200); 
  });
  
  wifiManager.setAPCallback([](WiFiManager* wm) {
    Serial.println("📶 Config Portal Started!");
    Serial.printf("🌐 Connect to: %s | Pass: %s\n", configPortalSsid.c_str(), CONFIG_PORTAL_PASS);
    Serial.println("🔗 Then visit: http://192.168.4.1");
    Serial.println("⚙️ You can also set MQTT Server and Port here");
    blinkLED(1, 1000);
  });
  
  Serial.println("🔗 Starting Config Portal (Timeout: 10 min)...");
  
  if (!wifiManager.autoConnect(configPortalSsid.c_str(), CONFIG_PORTAL_PASS)) {
    Serial.println("❌ Config Portal timed out or failed");
    blinkLED(5, 200);
    delay(2000);
    ESP.restart();
  }
  
  // ✅ اگر کاربر مقادیر MQTT را تغییر داد، ذخیره کن
  if (strlen(custom_mqtt_server.getValue()) > 0) {
    prefs.putString("mqtt_server", custom_mqtt_server.getValue());
    mqttServer = custom_mqtt_server.getValue();
    Serial.printf("✅ MQTT Server updated: %s\n", mqttServer.c_str());
  }
  if (strlen(custom_mqtt_port.getValue()) > 0) {
    int newPort = atoi(custom_mqtt_port.getValue());
    if (newPort > 0 && newPort <= 65535) {
      prefs.putInt("mqtt_port", newPort);
      mqttPort = newPort;
      Serial.printf("✅ MQTT Port updated: %d\n", mqttPort);
    }
  }
  
  Serial.println("✅ WiFi Connected via Config Portal!");
  Serial.printf("🌐 IP: %s | RSSI: %d dBm\n", 
                WiFi.localIP().toString().c_str(), WiFi.RSSI());
  blinkLED(2, 300);
}

// ==================== MQTT ====================
void initMQTT() {
  // ✅ استفاده از مقادیر پویا (از Preferences یا پیش‌فرض)
  mqttClient.setServer(mqttServer.c_str(), mqttPort);
  mqttClient.setCallback(onMqttMessage);
  mqttClient.setBufferSize(1024);
  
  #if MQTT_USE_SSL
    // اگر SSL فعال است، نیاز به WiFiClientSecure دارد
    // این بخش را فقط اگر سرور SSL دارد فعال کنید
    Serial.println("🔐 SSL mode enabled (requires WiFiClientSecure)");
  #endif
  
  Serial.printf("🔌 MQTT: %s:%d\n", mqttServer.c_str(), mqttPort);
}

void handleMQTT() {
  if (!mqttClient.connected()) {
    unsigned long now = millis();
    
    if (now - lastMqttReconnectAttempt > mqttReconnectDelay) {
      lastMqttReconnectAttempt = now;
      
      if (attemptMqttConnect()) {
        mqttReconnectDelay = MQTT_RECONNECT_MIN;
        mqttConnected = true;
        publishStatus(true);
        Serial.println("✅ MQTT Connected & Subscribed");
        blinkLED(2, 150);
      } else {
        mqttReconnectDelay = (mqttReconnectDelay * 2 < MQTT_RECONNECT_MAX) 
                              ? (mqttReconnectDelay * 2) 
                              : MQTT_RECONNECT_MAX;
        Serial.printf("⚠️ MQTT Connect failed. Next try in %lds\n", 
                      mqttReconnectDelay / 1000);
      }
    }
  } else {
    mqttClient.loop();
  }
}

bool attemptMqttConnect() {
  String clientId = "esp32-" + deviceId;
  String willTopic = "sh/" + deviceId + "/status";
  String willPayload = buildJson(deviceId.c_str(), channelStatus, false, isPaired);
  
  #if MQTT_USE_SSL
    if (mqttClient.connect(clientId.c_str(), MQTT_USER, MQTT_PASS, 
                           willTopic.c_str(), 1, true, willPayload.c_str())) {
  #else
    if (mqttClient.connect(clientId.c_str(), nullptr, nullptr, 
                           willTopic.c_str(), 1, true, willPayload.c_str())) {
  #endif
    
    // ✅ Subscribe به هر سه توپیک + بررسی جداگانه برای دیباگ
    String cmdTopic = "sh/" + deviceId + "/cmd";
    String unpairTopic = "sh/" + deviceId + "/unpair";
    String pairResponseTopic = "sh/" + deviceId + "/pair/response";
    
    bool sub1 = mqttClient.subscribe(cmdTopic.c_str(), 1);
    bool sub2 = mqttClient.subscribe(unpairTopic.c_str(), 1);
    bool sub3 = mqttClient.subscribe(pairResponseTopic.c_str(), 1);
    
    if (sub1 && sub2 && sub3) {
      Serial.printf("📡 Subscribed: cmd=%d, unpair=%d, pair/response=%d\n", sub1, sub2, sub3);
      Serial.printf("📡 Topics: %s, %s, %s\n", cmdTopic.c_str(), unpairTopic.c_str(), pairResponseTopic.c_str());
      return true;
    } else {
      Serial.printf("❌ Subscribe failed: cmd=%d, unpair=%d, pair/response=%d\n", sub1, sub2, sub3);
      mqttClient.disconnect();
      return false;
    }
  }
  
  Serial.printf("❌ MQTT connect failed, rc=%d\n", mqttClient.state());
  return false;
}

// ==================== پردازش پیام‌های MQTT ====================
void onMqttMessage(char* topic, byte* payload, unsigned int length) {
  String msg;
  msg.reserve(length + 1);
  for (unsigned int i = 0; i < length; i++) {
    msg += (char)payload[i];
  }
  
  Serial.printf("⚡ MQTT [%s]: %s\n", topic, msg.c_str());

  // ✅ ۱. پاسخ جفت‌سازی از سرور
  String topicStr(topic);
if (topicStr.endsWith("/pair/response")) {
  Serial.println("🔍 Received pair/response message");
  Serial.printf("📦 Raw message: %s\n", msg.c_str());  // ← لاگ کامل پیام
  
  // بررسی چند فرمت ممکن برای "success":true
  bool successFound = (msg.indexOf("\"success\":true") != -1) || 
                      (msg.indexOf("\"success\": true") != -1) ||
                      (msg.indexOf("\"success\":\"true\"") != -1);
  
  if (successFound) {
    Serial.println("✅ Pairing confirmed! Setting isPaired = true");
    isPaired = true;
    prefs.putBool("isPaired", true);  // ✅ ذخیره پایدار
    Serial.println("✅ Device is now SECURELY PAIRED (saved to Preferences)");
    blinkLED(2, 200);
    return;
  } else {
    Serial.println("❌ Pairing response did not contain success:true");
  }
}

  // ✅ ۲. فرمان Unpair (حذف دستگاه از اپ)
  if (topicStr.endsWith("/unpair")) {
    if (msg.indexOf("\"action\":\"unpair\"") != -1) {
      Serial.println("🔓 Unpair command received! Clearing pairing data...");
      factoryReset();
      return;
    }
  }

  // 🔒 ۳. امنیت: اگر جفت‌نشده، فقط Pairing را بپذیر
  bool isPairingRequest = (msg.indexOf("\"action\":\"pairing\"") != -1);
  if (!isPaired && !isPairingRequest) {
    Serial.println("⚠️ Device not paired. Command ignored for security");
    return;
  }
  
  if (isPairingRequest) {
    sendPairingRequest();
    return;
  }
  
  // ✅ ۴. اجرای فرمان معمولی
  String channelStr = extractField(msg, "channel");
  String stateStr = extractField(msg, "state");
  
  int channelIndex = channelStr.toInt();
  bool state = (stateStr.equalsIgnoreCase("true") || stateStr == "1");
  
  if (channelIndex >= 0 && channelIndex < NUM_CHANNELS) {
    executeCommandWithLog(channelIndex, state);
  } else {
    Serial.printf("⚠️ Invalid channelIndex: %d (max: %d)\n", 
                  channelIndex, NUM_CHANNELS - 1);
  }
}

// ==================== اجرای فرمان + لاگ توصیفی ====================
void executeCommandWithLog(int channelIndex, bool state) {
  int physicalPin = CHANNEL_PINS[channelIndex];
  
  digitalWrite(physicalPin, state ? HIGH : LOW);
  channelStatus[channelIndex] = state;
  
  Serial.printf("🔌 Channel %d (GPIO%d) → %s [via MQTT]\n", 
                channelIndex + 1, physicalPin, state ? "ON" : "OFF");
  
  publishStatus(true);
  blinkLED(1, 100);
}

// ==================== ساخت JSON وضعیت ====================
String buildJson(const char* devId, bool* channels, bool online, bool paired) {
  String json = "{";
  json += "\"deviceId\":\"" + String(devId) + "\",";
  
  for (int i = 0; i < NUM_CHANNELS; i++) {
    json += "\"ch" + String(i + 1) + "\":" + 
            String(channels[i] ? "true" : "false") + ",";
  }
  
  json += "\"online\":" + String(online ? "true" : "false") + ",";
  json += "\"paired\":" + String(paired ? "true" : "false") + ",";
  json += "\"timestamp\":" + String(millis());
  json += "}";
  return json;
}

// ==================== ساخت JSON درخواست Pairing ====================
String buildPairingJson() {
  String json = "{";
  json += "\"deviceId\":\"" + deviceId + "\",";
  json += "\"action\":\"pairing\",";
  json += "\"mac\":\"" + WiFi.macAddress() + "\",";
  json += "\"firmware\":\"1.0.0\",";
  json += "\"channels\":" + String(NUM_CHANNELS);
  json += "}";
  return json;
}

// ==================== ارسال وضعیت به سرور ====================
void publishStatus(bool force) {
  if (!force && millis() - lastStatusPublish < STATUS_PUBLISH_INTERVAL) {
    return;
  }
  
  String payload = buildJson(deviceId.c_str(), channelStatus, 
                            (WiFi.status() == WL_CONNECTED), isPaired);
  String statusTopic = "sh/" + deviceId + "/status";
  
  if (mqttClient.publish(statusTopic.c_str(), payload.c_str(), true)) {
    Serial.printf("📤 Status published to %s\n", statusTopic.c_str());
    if (force) lastStatusPublish = millis();
  } else {
    Serial.println("❌ Failed to publish status");
  }
}

// ==================== ارسال درخواست Pairing ====================
void sendPairingRequest() {
  String payload = buildPairingJson();
  String pairTopic = "sh/" + deviceId + "/pair";
  
  if (mqttClient.publish(pairTopic.c_str(), payload.c_str(), false)) {
    Serial.printf("🔗 Pairing request sent to %s\n", pairTopic.c_str());
    blinkLED(3, 200);
  } else {
    Serial.println("❌ Failed to send pairing request");
  }
}

// ==================== استخراج فیلد از JSON ====================
String extractField(const String& json, const String& key) {
  String search = "\"" + key + "\":";
  int start = json.indexOf(search);
  if (start == -1) return "";
  
  start += search.length();
  int end = json.indexOf(',', start);
  if (end == -1) end = json.indexOf('}', start);
  if (end == -1) end = json.length();
  
  String result = json.substring(start, end);
  result.trim();
  return result;
}

// ==================== مدیریت تاچ: Pairing و Factory Reset ====================
void handleTouch() {
  int reading = digitalRead(TOUCH_PIN);
  
  if (reading != lastTouchRead) {
    lastTouchChange = millis();
    lastTouchRead = reading;
  }
  if ((millis() - lastTouchChange) < TOUCH_DEBOUNCE_MS) {
    return;
  }
  
  // ✅ شروع فشار دادن
  if (reading == LOW && touchState == IDLE) {
    touchStartTime = millis();
    touchState = HOLDING;
    Serial.println("👆 Touch started...");
  }
  
  // ✅ در حال نگه‌داشتن
  else if (reading == LOW && touchState == HOLDING) {
    unsigned long duration = millis() - touchStartTime;
    
    if (duration < PAIRING_MIN_MS) {
      // کمتر از ۸ ثانیه: هیچ کاری نکن
    }
    else if (duration >= PAIRING_MIN_MS && duration < PAIRING_MAX_MS) {
      // ۸-۱۲ ثانیه: آماده‌ی Pairing (فیدبک اختیاری)
      if (duration % 1000 < 500) digitalWrite(LED_STATUS, HIGH);
      else digitalWrite(LED_STATUS, LOW);
    }
    else if (duration >= PAIRING_MAX_MS && duration < RESET_HOLD_MS) {
      // ۱۲-۲۰ ثانیه: Safe Zone - رها کن تا لغو شود
      if (duration % 500 < 250) digitalWrite(LED_STATUS, HIGH);
      else digitalWrite(LED_STATUS, LOW);
    }
    else if (duration >= RESET_HOLD_MS) {
      // بیش از ۲۰ ثانیه: ✅ Factory Reset!
      Serial.println("\n🔥 FACTORY RESET CONFIRMED (Hold >20s)!");
      blinkLED(5, 100);
      delay(500);
      factoryReset();
    }
  }
  
  // ✅ رها کردن دکمه
  else if (reading == HIGH && touchState == HOLDING) {
    unsigned long duration = millis() - touchStartTime;
    
    if (duration >= PAIRING_MIN_MS && duration < PAIRING_MAX_MS) {
      Serial.println("\n🔗 Pairing Mode Activated (8-12s hold)");
      sendPairingRequest();
    }
    else if (duration < PAIRING_MIN_MS) {
      Serial.printf("⏱️ Touch held for %lds (too short, ignored)\n", duration / 1000);
    }
    else if (duration >= PAIRING_MAX_MS && duration < RESET_HOLD_MS) {
      Serial.printf("⏱️ Touch held for %lds (safe zone, ignored)\n", duration / 1000);
    }
    
    touchState = IDLE;
    digitalWrite(LED_STATUS, LOW);  // خاموش کردن LED پس از رها کردن
  }
}

// ==================== Factory Reset ====================
void factoryReset() {
  Serial.println("🧹 Performing Factory Reset...");
  
  // ✅ پاک کردن فقط داده‌های مالکیت و شبکه
  prefs.remove("pairedUserId");
  prefs.remove("isPaired");
  prefs.remove("wifi_ssid");
  prefs.remove("wifi_pass");
  // ❌ deviceId پاک نمی‌شود (پایدار بر پایه‌ی مک)
  
  // پاک کردن تنظیمات WiFiManager
  WiFiManager wm;
  wm.resetSettings();
  
  Serial.println("✅ Ownership cleared. Device ID remains: " + deviceId);
  Serial.println("🔄 Restarting...");
  delay(1000);
  ESP.restart();
}

// ==================== توابع کمکی ====================
void blinkLED(int times, int duration) {
  for (int i = 0; i < times; i++) {
    digitalWrite(LED_STATUS, HIGH);
    delay(duration);
    digitalWrite(LED_STATUS, LOW);
    if (i < times - 1) delay(duration);
  }
}
